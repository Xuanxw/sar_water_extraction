cmake_minimum_required(VERSION 3.20)
project(sar_geometry_core LANGUAGES CXX)
add_library(sar_profile_regrid SHARED profile_regrid.cpp)
target_compile_features(sar_profile_regrid PRIVATE cxx_std_17)
set_target_properties(sar_profile_regrid PROPERTIES PREFIX "")
if(MINGW)
  target_link_options(sar_profile_regrid PRIVATE -static-libgcc -static-libstdc++)
endif()
add_executable(sar_geometry_core main.cpp geometry.cpp visibility.cpp gc_map2_v57.cpp)
target_compile_features(sar_geometry_core PRIVATE cxx_std_20)
find_package(OpenMP REQUIRED)
target_link_libraries(sar_geometry_core PRIVATE OpenMP::OpenMP_CXX)
if(MSVC)
  target_compile_options(sar_geometry_core PRIVATE /W4 /utf-8)
else()
  target_compile_options(sar_geometry_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(MINGW)
  target_link_options(sar_geometry_core PRIVATE -static -static-libgcc -static-libstdc++)
endif()
